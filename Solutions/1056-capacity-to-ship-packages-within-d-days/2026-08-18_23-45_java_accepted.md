## Solution

```java
class Solution {
    public boolean possible(int[] weights, int days,int k)
    {
        int temp=0;
        for(int i=0;i<weights.length;i++)
        {
            int val=weights[i];
            if(val>k)
            return false;
            if(temp+val>k)
            {
                days-=1;
                temp=0;
            }
            temp+=val;
            if(days==0)
            {
                return false;
            }
        }
        return true;
    }
    public int shipWithinDays(int[] weights, int days) {
        int l=1,h=Integer.MAX_VALUE;
        while(l<=h)
        {
            int mid=l+(h-l)/2;
            if(possible(weights,days,mid))
            h=mid-1;
            else
            l=mid+1;
        }
        return l;
    }
}
```
