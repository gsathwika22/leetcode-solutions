## Solution

```java
class Solution {
    public int calPoints(String[] nums) {
        Stack<Integer> st=new Stack<>();

        for(int i=0;i<nums.length;i++)
        {
            if(nums[i].equals("+"))
            {
            int top=st.pop();
            int top2=st.peek();
            int s=top+top2;
            
            st.push(top);
            st.push(s);
            }
            else if(nums[i].equals("D"))
            {
                st.push(2*st.peek());
            }
            else if(nums[i].equals("C"))
            {
                st.pop();
            }
            else{
                st.push(Integer.parseInt(nums[i]));
            }
            
        }
        int sum=0;
        for(int n:st)
        {
            sum+=n;
        }
        return sum;
    }
}
```
