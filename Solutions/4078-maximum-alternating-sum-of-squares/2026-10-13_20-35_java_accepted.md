## Solution

```java
import java.util.Arrays;

class Solution {
    public long maxAlternatingSum(int[] nums) {
    
        long[] sq=new long[nums.length];

        for(int i=0;i<nums.length;i++)
        sq[i]=(long)nums[i]*nums[i];

        Arrays.sort(sq);
        long ans = 0;
        int n = nums.length;
        int mid=n/2;

        for (int i = 0; i < mid; i++) {
         ans-=sq[i];

        }
        for(int i=mid;i<n;i++)
        ans+=sq[i];

        return ans;
    }
}
```
