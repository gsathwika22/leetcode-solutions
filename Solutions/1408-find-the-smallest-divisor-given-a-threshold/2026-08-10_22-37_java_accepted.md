## Solution

```java
class Solution {
  public static boolean isPossible(int[] nums, int threshold,int k)
    {
        int temp=0;
         for(int i=0;i<nums.length;i++)
        {
            int c=nums[i]/k;
            if(nums[i]%k!=0)
            c++;
            temp+=c;
        }

        if(temp<=threshold)
        {
            return true;
        }
        return false;
    }
    public int smallestDivisor(int[] nums, int threshold) {
       int l=1,h=(int)Math.pow(10,6);
       while(l<=h)
       {
        int mid=l+(h-l)/2;
        if(isPossible( nums,threshold,mid))
        {
            h=mid-1;
        }
        else
        l=mid+1;
       }
       return l;
    }
}
```
