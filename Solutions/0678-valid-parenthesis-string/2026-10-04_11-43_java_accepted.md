class Solution {
    public boolean checkValidString(String s) {

         int c=0,d=0,e=0,ans=0;
         for(int i=0;i<s.length();i++)
         {
            char ch=s.charAt(i);
            if(ch=='(')
            {
                c++;
                d++;
              
            }
            else if(ch==')')
            {
                c--;
               d--;
            }
            else
            {
                c--;
                d++;
            }
            if(d<0)
            return false;
         if(c<0)
         c=0;

         }  
         return c==0; 
    }
}
