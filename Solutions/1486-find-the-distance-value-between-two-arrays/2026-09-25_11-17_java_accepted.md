## Solution

```java
class Solution {
    public int findTheDistanceValue(int[] arr1, int[] arr2, int d)
     {
       Arrays.sort(arr2);
       int c=0;
       for(int i:arr1)
       {
        if(valid(arr2,i,d))
        c++;
       }
       return c;
    }
    public static boolean valid(int a[],int t,int d)
    {
        int l=0;
        int h=a.length-1;
        while(l<=h)
        {
            int mid=l+(h-l)/2;
            if(Math.abs(a[mid]-t)<=d)
            {
                return false;
            }
            else if(a[mid]<t)
            l=mid+1;
            else
            h=mid-1;
        }
        return true;
    }
}
```
