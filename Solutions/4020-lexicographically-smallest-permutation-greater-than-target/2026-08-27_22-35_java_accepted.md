## Solution

```java
class Solution {
    public String lexGreaterPermutation(String s, String target) {
        int n = s.length(), count[] = new int[26], pref[] = new int[26], max = 0;
        for (char c : s.toCharArray()) count[c - 'a']++;
        
        System.arraycopy(count, 0, pref, 0, 26);
        while (max < n && pref[target.charAt(max) - 'a']-- > 0) max++;

        for (int i = max; i >= 0; i--) {
            int[] rem = count.clone();
            for (int j = 0; j < i; j++) rem[target.charAt(j) - 'a']--;

            if (i < n) {
                for (int c = target.charAt(i) - 'a' + 1; c < 26; c++) {
                    if (rem[c] > 0) {
                        StringBuilder sb = new StringBuilder(target.substring(0, i)).append((char)('a' + c));
                        rem[c]--;
                        for (int k = 0; k < 26; k++) 
                            while (rem[k]-- > 0) sb.append((char)('a' + k));
                        return sb.toString();
                    }
                }
            }
        }
        return "";
    }
}
```
