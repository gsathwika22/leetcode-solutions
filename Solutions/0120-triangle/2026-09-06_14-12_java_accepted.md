class Solution {
    public int minimumTotal(List<List<Integer>> l) {
        int v1=0,v2=0;
        int n=l.size();
        int min=Integer.MAX_VALUE;
        int m=l.get(l.size()-1).size();
        int dp[][]=new int[n][m];
        dp[0][0]=l.get(0).get(0);
        for(int i=1;i<n;i++)
        {
            for(int j=0;j<l.get(i).size();j++)
            {
                    if(j == 0) {
                        dp[i][j] = dp[i-1][j] + l.get(i).get(j);
                    }
                    else if(j == l.get(i).size() - 1) {
                        dp[i][j] = dp[i-1][j-1] + l.get(i).get(j);
                    }
                    else {
                        dp[i][j] = Math.min(dp[i-1][j-1], dp[i-1][j])
                            + l.get(i).get(j);
                    }
            }
        }
        
      
        for(int j=0;j<m;j++)
        {
         
            min=Math.min(dp[n-1][j],min);
        }

      return min;

        }
}
