## Solution

```java
class Solution {
    public int minSumOfLengths(int[] arr, int t) {
       int n=arr.length;

       int min[]=new int[n];
       int l=0;
       int sum=0;
       int ans=Integer.MAX_VALUE;
       int b=Integer.MAX_VALUE;
       for(int r=0;r<n;r++)
       {
        sum+=arr[r];
        while(sum>t)
        sum-=arr[l++];

        if(sum==t)
        {
       int cur=r-l+1; 

       if(l>0 && min[l-1]!=Integer.MAX_VALUE){
        ans=Math.min(ans,cur+min[l-1]);
       }
       b=Math.min(b,cur);
       }
        min[r]=b;
       }
       return ans==Integer.MAX_VALUE?-1:ans;
    }
}
```
